Load the annotated data

# Load your annotated data
annotated_data <- read.csv("Silencing/GPS2/results/annotation/sictl_vs_sigps2_annotated.txt", sep = '\t')  # Replace with your file path

MSIGDB Analysis

# Assuming your data is in a dataframe called 'results'
# Extract the gene symbols
gene_list <- unique(annotated_data$SYMBOL)

# Remove any NA values
gene_list <- gene_list[!is.na(gene_list)]

# Get the desired MSigDB gene sets for mouse
msigdb_hallmark <- msigdbr(species = "Mus musculus", category = "H")

ora_result <- enricher(
  gene = gene_list,
  TERM2GENE = msigdb_hallmark %>% dplyr::select(gs_name, gene_symbol),
  pvalueCutoff = 0.05,
  pAdjustMethod = "BH"
)

#Choose the appropriate directory to save the file
output_dir <- "/projectnb/perissilab/Jawahar/GPS2_ChIPseq_Perissi-Lab/Silencing/GPS2/results/GSEA"

# Create the directory if it does not exist
if (!dir.exists(output_dir)) {
  dir.create(output_dir, recursive = TRUE)}

msigdb_result <- ora_result@result

output_file <- file.path(output_dir, paste0(basename("sictl_vs_sigps2"), "_msigdb.txt"))
write.table(as.data.frame(msigdb_result), file=output_file, sep="\t", quote=FALSE, row.names=FALSE)
